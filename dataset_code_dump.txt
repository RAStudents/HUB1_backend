@router.get("/yandex/callback", summary="РљРѕР»Р±СЌРє РЇРЅРґРµРєСЃ OAuth")
async def yandex_callback(
    code: str,
    service: AuthService = Depends(get_auth_service),
):
    _check_yandex_configured()

    async with httpx.AsyncClient() as client:
        token_resp = await client.post(
            "https://oauth.yandex.ru/token",
            data={
                "grant_type": "authorization_code",
                "code": code,
                "client_id": YANDEX_CLIENT_ID,
                "client_secret": YANDEX_CLIENT_SECRET,
            },
            headers={"Content-Type": "application/x-www-form-urlencoded"},
        )
        if token_resp.status_code != 200:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"РћС€РёР±РєР° РѕР±РјРµРЅР° РєРѕРґР° РЇРЅРґРµРєСЃ: {token_resp.text}",
            )
        yandex_token = token_resp.json().get("access_token")

        info_resp = await client.get(
            "https://login.yandex.ru/info?format=json",
            headers={"Authorization": f"OAuth {yandex_token}"},
        )
        if info_resp.status_code != 200:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="РќРµ СѓРґР°Р»РѕСЃСЊ РїРѕР»СѓС‡РёС‚СЊ РїСЂРѕС„РёР»СЊ РїРѕР»СЊР·РѕРІР°С‚РµР»СЏ РёР· РЇРЅРґРµРєСЃР°",
            )
        info = info_resp.json()

    yandex_id = str(info.get("id"))
    email = info.get("default_email") or f"{info.get('login')}@yandex.ru"
    name = info.get("real_name") or info.get("display_name") or info.get("login") or "Yandex User"

    return service.login_with_yandex(yandex_id=yandex_id, email=email, name=name)

@router.post("/complete-upload", response_model=SampleCardDTO)
def complete_upload(
    dataset_id: int,
    req: CompleteUploadRequest,
    user: CurrentUser = Depends(require_dataset_role({"owner", "labeler"})),
    dataset=Depends(get_dataset_or_404),
    s3: S3Client = Depends(get_s3_client),
    sample_repo=Depends(get_sample_repo),
) -> SampleCardDTO:
    """
    Р—Р°РІРµСЂС€РµРЅРёРµ Р·Р°РіСЂСѓР·РєРё РѕРґРёРЅРѕС‡РЅРѕРіРѕ РєР°РґСЂР° (`upload_service.py` Р»РѕРіРёРєР°):
    РґРѕРіСЂСѓР¶Р°РµС‚ Р±Р°Р№С‚С‹ РёР· _incoming, РєРѕРЅРІРµСЂС‚РёСЂСѓРµС‚ (heic/webp -> jpg), СЃРѕР·РґР°С‘С‚ РїСѓСЃС‚РѕР№ .txt,
    С‚СѓРјР±Сѓ Рё Р·Р°РІРѕРґРёС‚ СЃСЌРјРїР» РІ Р‘Р”.
    """
    raw_bytes = s3.get_bytes(req.upload_key)
    if not raw_bytes:
        raise HTTPException(status_code=404, detail="Р—Р°РіСЂСѓР¶РµРЅРЅС‹Р№ С„Р°Р№Р» РЅРµ РЅР°Р№РґРµРЅ РІ S3")

    orig_name = Path(req.upload_key).name
    split_name = req.split.lower()

    with scratch_file(suffix=Path(orig_name).suffix, data=raw_bytes) as tmp_src:
        img_bytes, norm_filename, content_type = normalize_image_file(tmp_src)
        stem = Path(norm_filename).stem

        with scratch_file(suffix=Path(norm_filename).suffix, data=img_bytes) as tmp_norm:
            thumb_bytes = render_thumbnail_bytes(
                image_path=tmp_norm,
                instances=[],
                task=TaskType(dataset.task),
            )

    s3_img_key = image_key(dataset_id, split_name, norm_filename)
    s3_lbl_key = label_key(dataset_id, split_name, stem)
    s3_thb_key = thumb_key(dataset_id, split_name, stem)

        dataset_id=dataset_id,
        stem=stem,
        split=split_name,
        image_key=s3_img_key,

        thumb_key=s3_thb_key,

        status=row.status,

class AnnotationService:

        self,

    def get_frame_annotations(
        self,
        dataset_id: int,
        sample_id: int,
        user_id: int,
        task: TaskType,
    ) -> FrameAnnotationsDTO:
        """
        РћС‚РґР°С‘С‚ СЂР°Р·РјРµС‚РєСѓ РєР°РґСЂР° С„СЂРѕРЅС‚РµРЅРґСѓ:
        РїСЂРѕРІРµСЂСЏРµС‚ РґРѕСЃС‚СѓРї -> РєР°С‡Р°РµС‚ .txt РёР· S3 -> РїР°СЂСЃРёС‚ С‡РµСЂРµР· cv -> РІРѕР·РІСЂР°С‰Р°РµС‚ DTO (РІ РіСЂР°РґСѓСЃР°С…).
        """
        # 1. РџСЂРѕРІРµСЂСЏРµРј, С‡С‚Рѕ РїРѕР»СЊР·РѕРІР°С‚РµР»СЊ СЃРѕСЃС‚РѕРёС‚ РІ РґР°С‚Р°СЃРµС‚Рµ (Р»СЋР±Р°СЏ СЂРѕР»СЊ: owner/labeler/observer)
        role = self.memberships.get_user_role(dataset_id=dataset_id, user_id=user_id)
        if role is None:
            raise PermissionError("Р”РѕСЃС‚СѓРї Рє РґР°С‚Р°СЃРµС‚Сѓ Р·Р°РїСЂРµС‰С‘РЅ")

        sample = self.samples.get_by_id(dataset_id=dataset_id, sample_id=sample_id)
        if sample is None:
            raise LookupError("РљР°РґСЂ РЅРµ РЅР°Р№РґРµРЅ")

        class_names = self.classes.get_names(dataset_id=dataset_id)
        s3_lbl_key = getattr(sample, "label_key", None) or label_key(
            dataset_id, sample.split, sample.stem
        )

        # 2. РЎРєР°С‡РёРІР°РµРј .txt РёР· S3 (РµСЃР»Рё С„Р°Р№Р»Р° РЅРµС‚ РёР»Рё РѕРЅ РїСѓСЃС‚РѕР№ вЂ” РѕС‚РґР°С‘Рј РїСѓСЃС‚РѕР№ СЃРїРёСЃРѕРє)
        raw_txt = self.s3.get_bytes(s3_lbl_key)

def detect_layout(root: Path) -> DatasetLayout:
    """
    РЈРіР°РґС‹РІР°РµС‚ СЂР°СЃРєР»Р°РґРєСѓ РґР°С‚Р°СЃРµС‚Р° (РґРІРµ СЂР°СЃРєР»Р°РґРєРё РёР· DATA_CONTRACT.md):
    - DIRECT: root / train / images  (РёР»Рё valid/images, test/images)
    - REVERSE: root / images / train
    """
    if (root / "images").is_dir() and any(
        (root / "images" / s).is_dir() for s in ("train", "val", "valid", "test")
    ):

    return DatasetLayout.DIRECT


def find_yaml_file(root: Path) -> Path | None:
    """РС‰РµС‚ РєРѕРЅС„РёРіСѓСЂР°С†РёРѕРЅРЅС‹Р№ yaml-С„Р°Р№Р» РІ РєРѕСЂРЅРµ РґР°С‚Р°СЃРµС‚Р°."""

        candidate = root / name
        if candidate.is_file():

    yaml_files = list(root.glob("*.yaml")) + list(root.glob("*.yml"))
    return yaml_files[0] if yaml_files else None


def build_acmenra_dataset(
    root: Path,

    fallback_names: List[str] | None = None,
) -> Dataset:
    """

    yaml_path = find_yaml_file(root)

    names = fallback_names or []

    if yaml_path is not None:

        yaml_names = raw_cfg.get("names")
        if isinstance(yaml_names, dict):

        names=names,

        val_leaf=val_leaf,

import hashlib
import secrets

def hash_password(password: str) -> str:
    salt = secrets.token_hex(16)
    pwd_hash = hashlib.sha256((password + salt).encode("utf-8")).hexdigest()
    return f"{salt}${pwd_hash}"

def verify_password(plain_password: str, hashed_password: str) -> bool:
    try:
        salt, pwd_hash = hashed_password.split("$")
        check_hash = hashlib.sha256((plain_password + salt).encode("utf-8")).hexdigest()
        return secrets.compare_digest(pwd_hash, check_hash)
    except Exception:
        return False

def create_access_token(user_id: int, expires_delta: timedelta = timedelta(days=7)) -> str:
    expire = datetime.now(timezone.utc) + expires_delta
    to_encode = {"sub": str(user_id), "exp": expire}

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])

        return int(user_id_str) if user_id_str else None
    except Exception:
        return None

def parse_label_file(
    path: Path,
    class_names: Sequence[str],
    task: TaskType,
    is_strict: bool = True,
) -> List[Instance]:
    """
    Р§РёС‚Р°РµС‚ YOLO .txt С„Р°Р№Р» Рё РІРѕР·РІСЂР°С‰Р°РµС‚ СЃРїРёСЃРѕРє Instance.
    РџСѓСЃС‚РѕР№ С„Р°Р№Р» РёР»Рё РѕС‚СЃСѓС‚СЃС‚РІРёРµ С„Р°Р№Р»Р° РєРѕСЂСЂРµРєС‚РЅРѕ РІРѕР·РІСЂР°С‰Р°РµС‚ [].
    """
    if not path.exists() or path.stat().st_size == 0:
        return []

    category = build_category_enum(class_names)
    nc = max(len(class_names), 1)
    parser = LabelParser(nc=nc, task_type=task, category=category)
    return parser.parse(path=path, is_strict=is_strict)

def instance_to_dto(instance: Instance, task: TaskType) -> AnnotationItemDTO:
    """
    РџРµСЂРµРІРѕРґРёС‚ Instance РёР· acmenra-cv РІ DTO РґР»СЏ API.
    РџР»РµР№СЃС…РѕР»РґРµСЂ-Р±РѕРєСЃ РґР»СЏ OBB Рё SEGMENT РЅР°СЂСѓР¶Сѓ РќР• РѕС‚РґР°С‘С‚СЃСЏ!
    РЈРіРѕР» OBB РїРµСЂРµРІРѕРґРёС‚СЃСЏ РёР· СЂР°РґРёР°РЅ РІ РіСЂР°РґСѓСЃС‹.
    """
    box_dto = None
    obb_dto = None
    polygon_dto = None

    if task == TaskType.DETECT:
        cx, cy, w, h = instance.box.to_xywhn()
        box_dto = BoxDTO(x=float(cx), y=float(cy), width=float(w), height=float(h))

    elif task == TaskType.OBB and instance.obb is not None:
        cx, cy, w, h, yaw_rad = instance.obb.to_xywhrn()
        obb_dto = ObbDTO(
            x=float(cx),
            y=float(cy),
            width=float(w),
            height=float(h),
            angle_deg=float(math.degrees(yaw_rad)),
        )

    elif task == TaskType.SEGMENT and instance.polygon is not None:
        pts = [PointDTO(x=float(p.X), y=float(p.Y)) for p in instance.polygon.points]
        polygon_dto = PolygonDTO(points=pts)

    return AnnotationItemDTO(
        id=instance.id,
        class_id=instance.class_id,
        box=box_dto,
        obb=obb_dto,
        polygon=polygon_dto,
    )


def dto_to_instance(
    dto: AnnotationItemDTO,
    task: TaskType,
    class_names: Sequence[str],
) -> Instance:
    """
    РЎРѕР±РёСЂР°РµС‚ Instance РёР· DTO С„СЂРѕРЅС‚РµРЅРґР° РґР»СЏ СЃРѕС…СЂР°РЅРµРЅРёСЏ РІ .txt Рё РѕС‚СЂРёСЃРѕРІРєРё Drawer-РѕРј.

    """
    category_enum = build_category_enum(class_names)
    members_by_val = {item.value: item for item in category_enum}
    label_enum = members_by_val.get(dto.class_id, list(category_enum)[0])

    # РџР»РµР№СЃС…РѕР»РґРµСЂ-Р±РѕРєСЃ РїРѕ СѓРјРѕР»С‡Р°РЅРёСЋ (РѕР±СЏР·Р°С‚РµР»РµРЅ РґР»СЏ Instance.__init__)
    placeholder_box = Box()

    if task == TaskType.DETECT and dto.box is not None:
        real_box = Box.from_xywhn([dto.box.x, dto.box.y, dto.box.width, dto.box.height])
        return Instance(
            id=dto.id,
            class_id=dto.class_id,
            label=label_enum,
            conf=1.0,
            box=real_box,
            obb=None,
            polygon=None,
        )

    if task == TaskType.OBB and dto.obb is not None:
        yaw_rad = math.radians(dto.obb.angle_deg)
        real_obb = Obb.from_xywhrn(
            [dto.obb.x, dto.obb.y, dto.obb.width, dto.obb.height, yaw_rad]
        )
        return Instance(
            id=dto.id,
            class_id=dto.class_id,
            label=label_enum,
            conf=1.0,
            box=placeholder_box,
            obb=real_obb,
            polygon=None,
        )

    if task == TaskType.SEGMENT and dto.polygon is not None:
        points = [Point.safe(x=pt.x, y=pt.y) for pt in dto.polygon.points]
        real_poly = Polygon(points=points)
        return Instance(
            id=dto.id,
            class_id=dto.class_id,
            label=label_enum,
            conf=1.0,
            box=placeholder_box,
            obb=None,
            polygon=real_poly,
        )

    raise ValueError(f"РќРµРєРѕСЂСЂРµРєС‚РЅС‹Рµ РґР°РЅРЅС‹Рµ РіРµРѕРјРµС‚СЂРёРё РґР»СЏ Р·Р°РґР°С‡Рё {task.value}")

KEEP_EXTENSIONS = {".jpg", ".jpeg", ".png"}
CONVERT_EXTENSIONS = {".heic", ".heif", ".webp", ".bmp", ".tiff", ".tif"}


def normalize_image_file(src_path: Path, jpeg_quality: int = 92) -> tuple[bytes, str, str]:
    """
    РџСЂРёРЅРёРјР°РµС‚ РїСѓС‚СЊ Рє РёСЃС…РѕРґРЅРѕРјСѓ С„Р°Р№Р»Сѓ РёР·РѕР±СЂР°Р¶РµРЅРёСЏ.

    """
    ext = src_path.suffix.lower()
    stem = src_path.stem

    if ext in KEEP_EXTENSIONS:
        content_type = "image/png" if ext == ".png" else "image/jpeg"
        return src_path.read_bytes(), f"{stem}{ext}", content_type

    if ext in CONVERT_EXTENSIONS:
        with Image.open(src_path) as img:
            rgb_img = img.convert("RGB")
            buf = BytesIO()
            rgb_img.save(buf, format="JPEG", quality=jpeg_quality)
            return buf.getvalue(), f"{stem}.jpg", "image/jpeg"

    raise ValueError(f"РќРµРїРѕРґРґРµСЂР¶РёРІР°РµРјС‹Р№ С„РѕСЂРјР°С‚ РёР·РѕР±СЂР°Р¶РµРЅРёСЏ: {ext}")

@contextmanager
def scratch_dir(prefix: str = "hub_") -> Iterator[Path]:
    """
    РЎРѕР·РґР°С‘С‚ РІСЂРµРјРµРЅРЅСѓСЋ РґРёСЂРµРєС‚РѕСЂРёСЋ (РЅР°РїСЂРёРјРµСЂ, РґР»СЏ СЂР°СЃРїР°РєРѕРІРєРё zip РёР»Рё СЃР±РѕСЂРєРё СЌРєСЃРїРѕСЂС‚Р°)

    """
    tmp_path = Path(tempfile.mkdtemp(prefix=prefix))
    try:
        yield tmp_path
    finally:
        shutil.rmtree(tmp_path, ignore_errors=True)


@contextmanager
def scratch_file(suffix: str = "", data: bytes | None = None) -> Iterator[Path]:
    """
    РЎРѕР·РґР°С‘С‚ РІСЂРµРјРµРЅРЅС‹Р№ С„Р°Р№Р» (РЅР°РїСЂРёРјРµСЂ, С‡С‚РѕР±С‹ СЃРєРѕСЂРјРёС‚СЊ Р±Р°Р№С‚С‹ РёР· S3 РІ LabelParser РёР»Рё ImageParser)
    Рё СѓРґР°Р»СЏРµС‚ РµРіРѕ СЃСЂР°Р·Сѓ РїРѕСЃР»Рµ Р·Р°РІРµСЂС€РµРЅРёСЏ СЂР°Р±РѕС‚С‹.
    """
    import os

    fd, raw_path = tempfile.mkstemp(suffix=suffix)
    os.close(fd)  # Р’Р°Р¶РЅРѕ РґР»СЏ Windows: СЃСЂР°Р·Сѓ РѕСЃРІРѕР±РѕР¶РґР°РµРј РґРµСЃРєСЂРёРїС‚РѕСЂ РћРЎ!

def extract_zip_safely(zip_bytes_or_path: bytes | Path, dest_dir: Path) -> Path:
    """
    Р Р°СЃРїР°РєРѕРІС‹РІР°РµС‚ Р°СЂС…РёРІ РІРѕ РІСЂРµРјРµРЅРЅСѓСЋ РїР°РїРєСѓ. Р—Р°С‰РёС‰Р°РµС‚ РѕС‚ Zip Slip (РІС‹С…РѕРґР° Р·Р° РїСЂРµРґРµР»С‹ РїР°РїРєРё).

    """
    if isinstance(zip_bytes_or_path, bytes):
        zip_file_path = dest_dir / "_upload.zip"
        zip_file_path.write_bytes(zip_bytes_or_path)
    else:
        zip_file_path = zip_bytes_or_path

    extract_root = dest_dir / "extracted"
    extract_root.mkdir(parents=True, exist_ok=True)

    with zipfile.ZipFile(zip_file_path, "r") as zf:
        for member in zf.infolist():
            # РРіРЅРѕСЂРёСЂСѓРµРј СЃРёСЃС‚РµРјРЅС‹Р№ РјСѓСЃРѕСЂ macOS

    if isinstance(zip_bytes_or_path, bytes):

    # Р•СЃР»Рё РІСЃС‘ СѓРїР°РєРѕРІР°РЅРѕ РІ РѕРґРЅСѓ РІР»РѕР¶РµРЅРЅСѓСЋ РїР°РїРєСѓ вЂ” СЃРїСѓСЃРєР°РµРјСЃСЏ РІ РЅРµС‘
    subdirs = [p for p in extract_root.iterdir() if not p.name.startswith(".")]
    if len(subdirs) == 1 and subdirs[0].is_dir():
        return subdirs[0]

    return extract_root

class DataBase:
    MAX_LEN = 100
    """
    A high-level asynchronous wrapper for managing PostgreSQL database connections with automatic recovery.

    This class provides:
    - Secure connection management with automatic reconnection on failures
    - Schema management and table caching
    - ORM-like access to database tables
    - Comprehensive input validation to prevent SQL injection
    - Automatic loading of tables and their primary keys on initialization

    Important: This is an async-first implementation. All database operations must be awaited.
    The class automatically handles connection recovery, but application code should be designed
    to handle potential transient errors during high-load scenarios.
    """

    def __init__(self, dbname: str, user: str, password: str, host: str = "localhost", port: int = 5432):
        """
        Initialize the database wrapper without establishing a connection.

        Note: This constructor only stores connection parameters. No network operations are
        performed at this stage. To establish a connection, call the async connect() method.

        Args:
            dbname: Name of the database to connect to.
            user: Username for authentication (must be valid PostgreSQL identifier).
            password: Password for authentication.
            host: Database host address (default: localhost).
            port: Database port number (default: 5432).

        Raises:
            TypeError: If any parameter has incorrect type.
            ValueError: If any parameter fails validation (length, format).

        Security Note: Passwords are stored in memory but not logged. Ensure your environment
        protects process memory from unauthorized access.
        """
        self.dbname = dbname
        self.user = user
        self.password = password
        self.host = host
        self.port = port
        self._connection = None
        self._tables = None

    def __len__(self) -> int:
        """
        Get the number of tables in the database.

        This method returns the count of tables that have been loaded into the internal cache
        during initialization or connection recovery.

        Returns:
            Number of tables in the 'public' schema that are currently cached.
        """
        return len(self.table_names)

    def __str__(self):
        """
        Return a human-readable overview of the database.

        Returns:
            Pretty-formatted string with table names, row counts, and status.
        """
        table = PrettyTable()
        table.title = f"{'Empty ' if not len(self.table_names) > 0 else ''}DataBase"
        table.field_names = ["Name", "Type", "Columns count", "Rows count", "Status"]
        for table_name in self.table_names:
            loop = asyncio.new_event_loop()
            this_table = loop.run_until_complete(self.get_table(table_name))
            db_status = this_table.status
            db_rows_cnt = this_table.count
            db_columns_cnt = this_table.columns_count
            if db_status == Status.ENABLE:
                status = DataBase._set_str_cmd_clr("ENABLE", 'GREEN')
            else:
                status = DataBase._set_str_cmd_clr("EMPTY", 'RED')
            table.add_row([table_name, "<DataBase>", db_columns_cnt, db_rows_cnt, status])
        return str(table)

    async def create_database(self, dbname: str, owner: Optional[str] = None) -> bool:
        """
        Create a new database if it does not exist.

        This method connects to the 'postgres' system database to execute the CREATE DATABASE
        command. It first checks if the database already exists to avoid errors.

        Args:
            dbname: Name for the new database (must be valid identifier).
            owner: Optional role name to own the database.

        Returns:
            True if database was created or already exists, False otherwise.

        Raises:
            TypeError: If parameters have incorrect types.
            ValueError: If parameters fail validation.
            ConnectionError: If unable to connect to system database.

        Security Note: This operation requires superuser privileges or appropriate
        CREATEDB permissions. The method safely handles identifier quoting to prevent
        SQL injection.
        """
        if not isinstance(dbname, str):
            raise TypeError(f"Field 'dbname' should be '<str>', but got {type(dbname).__name__}")
        if not 0 < len(dbname) <= self.MAX_LEN:
            raise ValueError(f"'dbname' length should be in range [0, {self.MAX_LEN}], but got '{len(dbname)}'")

        if owner is not None and not isinstance(owner, str):
            raise TypeError(f"'owner' must be '<str>', but got {type(owner).__name__}")
        if owner is not None and not 0 < len(owner) <= self.MAX_LEN:
            raise ValueError(f"'owner' length should be in range [0, {self.MAX_LEN}], but got '{len(owner)}'")

        if await self._database_exists(dbname):
            logger.info(f"Database '{dbname}' already exists.")
            return True

        query = sql.SQL("CREATE DATABASE {dbname} OWNER {owner}") if owner else sql.SQL("CREATE DATABASE {dbname}")
        query = query.format(dbname=sql.Identifier(dbname), owner=sql.Identifier(owner) if owner else None)
        logger.info(f"[SQL] Create database...")
        if logger.getEffectiveLevel() <= logging.DEBUG:
            logger.debug(f"[SQL] Create database '{self.dbname}' with owner: '{owner}'")
        return await self.execute_admin_command(query)

    async def create_role(self, role_name: str, password: Optional[str] = None, superuser: bool = False) -> None:
        """
        Create a new database role (user account) with specified privileges.

        This method safely creates a role using parameterized queries to prevent SQL injection.
        The password is properly quoted using PostgreSQL's quote_literal function.

        Args:
            role_name: Name of the role to create (must be valid identifier).
            password: Optional password for the role.
            superuser: Whether to grant SUPERUSER privileges.

        Raises:
            TypeError: If parameters have incorrect types.
            ValueError: If parameters fail validation.
            ConnectionError: If unable to connect to system database.
            InsufficientPrivilege: If current user lacks CREATE ROLE permission.

        Security Note: Creating superusers should be done with extreme caution. This
        method validates all inputs and uses secure quoting mechanisms, but the calling
        code should implement additional authorization checks as appropriate for your
        security requirements.
        """
        await self._ensure_connection()

        if not isinstance(role_name, str):
            raise TypeError(f"Field 'role_name' should be '<str>', but got {type(role_name).__name__}")
        if not 0 < len(role_name) <= self.MAX_LEN:
            raise ValueError(f"'role_name' length should be in range [0, {self.MAX_LEN}], but got '{len(role_name)}'")

        if password is not None and not isinstance(password, str):
            raise TypeError(f"Field 'password' should be '<str>', but got {type(password).__name__}")
        if password is not None and not 0 < len(password) <= self.MAX_LEN:
            raise ValueError(f"'password' length should be in range [0, {self.MAX_LEN}], but got '{len(password)}'")

        if not isinstance(superuser, bool):
            raise TypeError(f"Field 'superuser' should be '<bool>', but got {type(password).__name__}")

        query_parts = [sql.SQL("CREATE ROLE {} LOGIN").format(sql.Identifier(role_name))]
        if password:
            escaped_password = await self._escape_string(password)
            query_parts.append(sql.SQL("PASSWORD {}").format(sql.Literal(escaped_password)))
        if superuser:
            query_parts.append(sql.SQL("SUPERUSER"))
        await self.execute_admin_command(sql.SQL(" ").join(query_parts))

    async def add_table(self,
                        table_name: str,
                        columns: Dict[str, DBType],
                        primary_key: Optional[str] = None,

        """
        Create a new table and add it to the internal cache.

        This method safely creates a table with proper identifier quoting to prevent

        is automatically created as the primary key.

            primary_key: Optional name of the primary key column.

            ValueError: If parameters fail validation.
            DatabaseError: If table creation fails at database level.

        if not isinstance(table_name, str):
            raise TypeError(f"Field 'table_name' should be '<str>', but got {type(table_name).__name__}")

            raise ValueError(f"'table_name' length should be in range [0, {self.MAX_LEN}], but got '{len(table_name)}'")

                raise TypeError(f"Field 'column' should be '<str>', but got {type(column).__name__}")

            if not isinstance(col_type, DBType):

            if not 0 < len(primary_key) <= self.MAX_LEN:
                raise ValueError(f"'primary_key' length should be in [0, {self.MAX_LEN}], but got '{len(primary_key)}'")

                    index_name=sql.Identifier(index_name),
                    table=sql.Identifier(table_name),
                    column=sql.Identifier(index_column)
                )

                async with self._connection.cursor() as cursor:

    async def delete_table(self, table_name: str) -> None:
        """
        Drop a table and remove it from the cache.

        This method executes a DROP TABLE command with CASCADE option to remove
        the table and all its dependent objects. It also updates the internal cache.

        Args:
            table_name: Name of the table to drop (must be valid identifier).

        Raises:
            TypeError: If table_name is not a string.
            ValueError: If table_name fails validation or table doesn't exist.
            DatabaseError: If DROP command fails at database level.

        Security Note: This operation requires appropriate DROP privileges. The method
        validates inputs and uses proper identifier quoting to prevent SQL injection.
        Use with caution as this operation is destructive and cannot be undone.
        """
        await self._ensure_connection()

        if not isinstance(table_name, str):
            raise TypeError(f"Field 'table_name' should be '<str>', but got {type(table_name).__name__}")
        if not 0 < len(table_name) <= self.MAX_LEN:

        if table_name not in self.table_names:

    async def connect(self):
        """
        Establish database connection and initialize metadata cache.

        This method performs the following steps:
        1. Creates the database if it doesn't exist
        2. Establishes connection to the target database
        3. Loads all tables and their metadata into cache

        Note: This method is automatically called by the create() factory method.
        Most applications should use create() rather than calling connect() directly.

        Raises:
            ConnectionError: If database connection cannot be established.
            DatabaseError: If metadata queries fail after connection.

        Security Note: Connection parameters are validated before use. SSL is enforced
        in production environments (when PG_SSL_CERT_PATH is set). The connection
        uses keepalive parameters to prevent timeouts during long-running operations.
        """
        await self.create_database(dbname=self.dbname, owner=self.user)
        await self._create_cursor()
        self._tables = await self._load_all_tables()

    async def _load_all_tables(self) -> Dict[str, Table]:
        """
        Internal method to load all tables from the 'public' schema into cache.

        This method queries the information_schema to discover all tables and their
        primary keys in the public schema. It should only be called during initialization
        or after connection recovery.

        Returns:
            Dictionary mapping table names to initialized Table objects.

        Raises:
            ConnectionError: If database connection is not active.
            DatabaseError: If metadata queries fail.

        Implementation Note: This is an internal method and should not be called directly
        by application code. It's used during initialization and connection recovery.
        The method assumes a valid connection exists and will raise errors if not.
        """
        await self._ensure_connection()

        query = """
            SELECT tc.table_name, kcu.column_name AS pk_column
            FROM information_schema.table_constraints tc
            LEFT JOIN information_schema.key_column_usage kcu 
                ON tc.constraint_name = kcu.constraint_name
            WHERE tc.constraint_type = 'PRIMARY KEY'
              AND tc.table_schema = 'public'
            ORDER BY tc.table_name;
        """

        async with self._connection.cursor() as cursor:
            await cursor.execute(query)
            tables = {}
            for table_name, pk_column in await cursor.fetchall():
                tables[table_name] = Table(connection=self._connection,
                                           table_name=table_name,
                                           primary_key=pk_column if pk_column else "id")
            return tables

    async def _get_tables_names(self) -> List[str]:
        """
        Internal method to get a list of all table names in the 'public' schema.

        This method queries the information_schema to discover table names.
        It's used internally by other methods and should not be called directly.

        Returns:
            List of table names in the 'public' schema.

        Raises:
            ConnectionError: If database connection is not active.
            DatabaseError: If metadata query fails.

        Implementation Note: This is an internal helper method. Prefer using
        table_names property for application code.
        """
        await self._ensure_connection()

        async with self._connection.cursor() as cursor:
            await cursor.execute("""
                SELECT table_name
                FROM information_schema.tables
                WHERE table_schema = 'public'
                ORDER BY table_name;
            """)
            return [row[0] for row in await cursor.fetchall()]

        This method creates a new connection with keepalive parameters to prevent

                await self._connection.close()
            self._connection = await psycopg.AsyncConnection.connect(
                dbname=self.dbname,
                user=self.user,
                password=self.password,
                host=self.host,
                port=self.port,
                # sslmode='verify-full',

def letterbox_h(image: np.ndarray, height: int = PREVIEW_H) -> np.ndarray:
    h, w = image.shape[:2]
    if h == height:
        return image
    scale = height / float(h)
    return cv2.resize(image, (max(1, int(round(w * scale))), height), interpolation=cv2.INTER_LINEAR)


def caption(image: np.ndarray, text: str) -> np.ndarray:
    out = image.copy()
    cv2.rectangle(out, (0, 0), (out.shape[1], 36), (0, 0, 0), -1)
    cv2.putText(out, text, (12, 26), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2, cv2.LINE_AA)
    return out


def draw(image: np.ndarray, instances, task: TaskType) -> np.ndarray:

    return Drawer(style=style).draw_instances(frame=frame,

                                              is_obb=task == TaskType.OBB,
                                              is_polygon=task == TaskType.SEGMENT)


def pair_canvas(original, augmented, left, right) -> np.ndarray:
    a = caption(letterbox_h(original), left)
    b = caption(letterbox_h(augmented), right)
    return np.hstack([a, b])


def preview_split(split, task: TaskType, title: str):
    if split is None or len(split) == 0:
        print(f"skip {title}: empty")
        return True
    idx = 0
    bar = tqdm(total=len(split), desc=f"aug/{title}")
    while 0 <= idx < len(split):
        sample = split[idx]
        image = split.image_parser.parse(path=sample.image_path, is_strict=False)
        if image is None:
            idx += 1
            bar.update(1)
            continue
        labels = split.get_instances(idx)
        aug_image, aug_labels = pipeline(image, labels)
        canvas = pair_canvas(
            draw(image, labels, task),
            draw(aug_image, aug_labels, task),
            f"{title}  {sample.stem}  n={len(labels)}",
            f"AUG  n={len(aug_labels)}  steps={len(pipeline)}",
        )
        cv2.imshow("Acmenra Augmentation  |  original LEFT  /  aug RIGHT", canvas)
        key = cv2.waitKey(WAIT_MS) & 0xFF
        if key == ord("q"):
            bar.close()
            return False
        if key == ord("r"):
            continue
        if key == ord("s"):
            SAVE_DIR.mkdir(parents=True, exist_ok=True)
            cv2.imwrite(str(SAVE_DIR / f"{title}_{sample.stem}.jpg"), canvas)
            continue
